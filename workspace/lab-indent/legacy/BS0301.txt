PROCEDURE Valid
*** service code generation

select max(code) from services ;
	 where dept = allt(this.value) into array tmpary
** code generation	 
if _tally =0
	xcode = allt(this.value) + '001'
else
	xv = allt(str(int(val(right(allt(tmpary(1)),3))) + 1))
	xcode = allt(this.value) + padl(xv,3,'0')	
Endif
this.parent.txtcode.value=xcode
this.parent.txtcode.refresh
*select name,id from sergrp where dept=allt(this.value) order by name into cursor cbogroup
ENDPROC

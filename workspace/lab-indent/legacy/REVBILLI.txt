select ipdbill.billno , ipdbill.hos_code, ipdbill.pcode, patient.name as pat, ipdbill.cid, company.name as com, ;
	   Ntot , " " as posted ;
  from ipdbill ;
  join register on (register.regno = ipdbill.hos_code and register.type = 'I') ;
  left outer join patient on (ipdbill.pcode=patient.code) ;
  left outer join company on (ipdbill.cid = company.id) ;
  where &lccond and empty(ipdbill.stat) ;
	order by ipdbill.billno ;  
	into table (tmpfil)
use
use invdet order regno in 0 
use (tmpfil) alia tmppost in 0 excl
sele tmppost
set rela to hos_code into invdet
scan
	if !eof('invdet')
		dele
	endif
ends
set safety on

	wait window 'Bills are not generated, No Records available'
	return 
endif

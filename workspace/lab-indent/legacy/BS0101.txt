PROCEDURE updbuttons
public nused
nused = 0
oldalias = select()
thisform.txtcntphar.value = ' 0:'
thisform.txtcntlab.value = ' 0:'
thisform.txtcntxray.value = ' 0:'
thisform.txtcntus.value = ' 0:'
thisform.txtcntden.value = ' 0:'
thisform.txtcntcnp.value = ' 0:'
select dept,count(slipno),sum(iif(type<>'R',ntotal,ntotal*-1)) as sum_ntotal;
	from obil_mst ;	
	where hos_code=register.regno group by dept into cursor cntslip

*select dept,count(slipno) from obil_mst where hos_code=register.regno group by dept into cursor cntslip

select cntslip
go top
do while !eof()
	nused = nused + cntslip.sum_ntotal
	do case
		case dept='10'
			thisform.txtcntphar.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
		case dept='01'
			thisform.txtcntlab.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
		case dept='02'
			thisform.txtcntxray.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
		case dept='03'
			thisform.txtcntus.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
		case dept='04'
			thisform.txtcntden.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
		case dept='11'
			thisform.txtcntcnp.value = str(cntslip.cnt_slipno,2)+':'+str(cntslip.sum_ntotal,7,2)
	endcase
	skip
enddo
thisform.baseframe1.page1.txtused.value = nused
use in cntslip
select (oldalias)
dodefault()

ENDPROC
PROCEDURE Init
dodefault()
thisform.chkadd = .F.
thisform.chkdelete = .F.
thisform.chkedit = .F.

ENDPROC

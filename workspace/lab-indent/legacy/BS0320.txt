	use compchg again
	set order to cidscode
	if seek(thisform.baseframe1.page1.cbocid.value+ ;
		thisform.baseframe1.page1.cboscode.value)
		use
		select (oldalias)
		thisform.validmsg = "Duplicate entry found, Re-enter"
		return .f.
	endif
	use
	select (oldalias)
endif

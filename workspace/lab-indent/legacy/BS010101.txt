KPROCEDURE updbuttons
dodefault()
local var1,var2
var1 = 'mregno'+alltrim(str(thisform.datasessionid))

		ctoolbar.cmdbottom.enabled = .f.
		ctoolbar.cmdtop.enabled = .f.
		ctoolbar.cmdnext.enabled = .f.
		ctoolbar.cmdprev.enabled = .f.

	ctoolbar.cmdedit.enabled = .f.
	ctoolbar.cmddelete.enabled = .f.
endif

PROCEDURE beforebottom
local var1,var2
var1 = 'mregno'+alltrim(str(thisform.datasessionid))
var2 = 'mdept'+alltrim(str(thisform.datasessionid))

do while obil_mst.dept+str(obil_mst.hos_code,7) = &var2+str(&var1,7) and !eof()
	skip

return .f.

ENDPROC
PROCEDURE beforenext
local var1,var2
var1 = 'mregno'+alltrim(str(thisform.datasessionid))
var2 = 'mdept'+alltrim(str(thisform.datasessionid))

IF !EOF()
	Skip 1
	IF obil_mst.dept+str(obil_mst.hos_code,7)<>&var2+str(&var1,7) or EOF()
		thisform.chkbottom = .t.
		skip -1
	ENDIF
ENDIF
return .f.

ENDPROC
PROCEDURE beforeprev
local var1,var2
var1 = 'mregno'+alltrim(str(thisform.datasessionid))
var2 = 'mdept'+alltrim(str(thisform.datasessionid))

	Skip -1
	IF obil_mst.dept+str(obil_mst.hos_code,7)<>&var2+str(&var1,7) or BOF()
		thisform.chktop = .t.
		skip
	ENDIF
ENDIF
return .f.

ENDPROC
PROCEDURE beforetop

	skip -1
enddo
skip
return .f.

ENDPROC

PROCEDURE cmdok.Click
select tmppost
go top

do while !eof()
	if posted <> 'X'
		skip
		loop
	endif
	** Generate Bill # **
	select bcontrol
	if rlock()
		Mbillno = bcontrol.opdbill + 1
		replace opdbill with Mbillno in bcontrol
	endif
	unlock

 	select OPDBILL
 	append blank in opdbill
 	replace billno with	Mbillno, ;
 			bdate with thisform.txtbdate.value, ;
 			sdate with thisform.txtsdate.value, ;
	 		edate with thisform.txtedate.value, ;
 			hos_code with tmppost.regno , ;
	 		pcode with tmppost.pcode, ;
	 		cid	 with tmppost.cid, ;
	 		ntot  with tmppost.ntot ;
	 		in opdbill
	if fsize("userid",'opdbill') = 10
		replace userid with gcloginid, ;
				udate with date(), ;
				utime with time() ;
				in opdbill
	endif

 	select obil_mst
	=seek(tmppost.regno, "obil_mst")
	do while Hos_code = tmppost.regno and !eof()
		if Bill = 0 and ;
			between(slipdt, thisform.txtsdate.value, thisform.txtedate.value) 
			replace bill with Mbillno , ;
				   bdate with thisform.txtbdate.value in obil_mst
		endif
		skip
	enddo
	select register
	seek tmppost.regno
	if !eof()
		replace billgen with .T., tbillamt with tmppost.ntot
	endif
	select tmppost
	skip
enddo           

select opdbill
=tableupdate(1,.t.)

select obil_mst
=tableupdate(1,.t.)

select register
=tableupdate(1,.t.)

wait "Bills successfully generated" window

thisform.release

ENDPROC

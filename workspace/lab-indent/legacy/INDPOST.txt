tPROCEDURE cmdok.Click
msgtype = 0
select 0
use ..\billsys\obil_mst
set order to ds
=CURSORSETPROP("Buffering", 5, "obil_mst")
select 0
use ..\billsys\obil_det
=CURSORSETPROP("Buffering", 5, "obil_det")

select 0
use ..\billsys\ibil_mst
set order to ds
=CURSORSETPROP("Buffering", 5, "ibil_mst")
select 0
use ..\billsys\ibil_det
=CURSORSETPROP("Buffering", 5, "ibil_det")

select tmppost
set relation to (left(type,1)+str(regno,7)) into register
go top
do while !eof()

	if posted = ' ' OR REGISTER.BILLGEN
		skip
		loop
	endif
	select collect
	seek tmppost.indentno
	if type='OPD'
		** Do not post if indent # already exists in billing
		if seek(gcdept+tmppost.indentno,"obil_mst")
			select tmppost
			skip
			msgtype = 2
			loop
		endi
		msgtype = iif(msgtype <> 2,1,2)
		**updating records in obil_mst.dbf from collect.dbf
		append blank in obil_mst
		replace ;
			dept   	 with gcdept, ;
			slipno 	 with collect.indentno, ; 
			hos_code with collect.regno, ;
			slipdt 	 with collect.date, ;
			type	 with 'A', ;
			ntotal	 with collect.total,;
			stat	 with 'P' in obil_mst
			
		**updating records in obil_det.dbf from trans.dbf
		select trans
		seek collect.indentno
		do while indentno = collect.indentno and !eof()
			if empty(sid)
				skip
				loop
			Endif
			append blank in obil_det
			replace ;
				dept 	with gcdept, ;
				slipno	with trans.indentno, ;
				code 	with trans.sid, ;
				qty		with 1, ;
				rate	with trans.charges, ;			
				namt	with trans.charges, ;
				rid		with trans.rid, ;
				flag	with 'B', ;
				concode with trans.concode, ;
				conpay  with iif(empty(trans.concode),0, ;
							iif(!seek(collect.cid+trans.concode+trans.sid,"concomm"),0, ;
								trans.charges-round((trans.charges*(concomm.hmcper/100)),2) )), ;
				desc	with trans.sname in obil_det
			
			select trans
			skip
		Enddo
	else
		** Do not post if indent # already exists in billing
		if seek(gcdept+tmppost.indentno,"ibil_mst")
			select tmppost
			skip
			msgtype = 2
			loop
		endi
		msgtype = iif(msgtype <> 2,1,2)
		**updating records in ibil_mst.dbf from collect.dbf
		append blank in ibil_mst
		replace ;
			dept   	 with gcdept, ;
			slipno 	 with collect.indentno, ; 
			hos_code with collect.regno, ;
			slipdt 	 with collect.date, ;
			type	 with 'A', ;
			ntotal	 with collect.total,;
			stat	 with 'P' in ibil_mst
			
		**updating records in ibil_det.dbf from trans.dbf
		select trans
		seek collect.indentno
		do while indentno = collect.indentno and !eof()
			if empty(sid)
				skip
				loop
			Endif
			append blank in ibil_det
			replace ;
				dept 	with gcdept, ;
				slipno	with trans.indentno, ;
				code 	with trans.sid, ;
				qty		with 1, ;
				rate	with trans.charges, ;			
				namt	with trans.charges, ;
				rid		with trans.rid, ;
				flag	with 'B', ;
				concode with trans.concode, ;
				conpay  with iif(empty(trans.concode),0, ;
							iif(!seek(collect.cid+trans.concode+trans.sid,"concomm"),0, ;
								trans.charges-round((trans.charges*(concomm.hmcper/100)),2) )), ;
				desc	with trans.sname in ibil_det
			
			select trans
			skip
		Enddo
	endif

	select collect
	replace posted with 'P'
		
	select tmppost
	skip
Enddo
select collect
=tableupdate(1,.t.)

select obil_mst
=tableupdate(1,.t.)
use

select obil_det
=tableupdate(1,.t.)
use

select ibil_mst
=tableupdate(1,.t.)
use

select ibil_det
=tableupdate(1,.t.)
use
do case 
case msgtype = 1
	wait wind nowait "Record Successfully Posted"
case msgtype = 2
	wait wind nowait "Some indents may not have been posted"
endcase

thisform.release

ENDPROC

		left(type,1) $ iif(thisform.opttype.value=3,'OI', ;

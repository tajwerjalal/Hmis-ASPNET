vPROCEDURE cmdok.Click
select 0
use ..\billsys\obil_mst
=CURSORSETPROP("Buffering", 5, "obil_mst")
select 0
use ..\billsys\obil_det
=CURSORSETPROP("Buffering", 5, "obil_det")

select tmppost
go top
do while !eof()

		if posted = ' '
			skip
			loop
		endif
		select collect
		seek tmppost.indentno
		**updating records in obil_mst.dbf from collect.dbf
		append blank in obil_mst
		replace ;
			dept   	 with gcdept, ;
			slipno 	 with collect.indentno, ; 
			hos_code with collect.regno, ;
			slipdt 	 with collect.date, ;
			type	 with 'A', ;
			ntotal	 with collect.total,;
			stat	 with 'P' in obil_mst
			
		**updating records in obil_det.dbf from trans.dbf
		select trans
		seek collect.indentno
		do while indentno = collect.indentno and !eof()
			if empty(sid)
				skip
				loop
			Endif
			append blank in obil_det
			replace ;
				dept 	with gcdept, ;
				slipno	with trans.indentno, ;
				code 	with trans.sid, ;
				qty		with 1, ;
				rate	with trans.charges, ;			
				namt	with trans.charges, ;
				rid		with trans.rid, ;
				desc	with trans.sname in obil_det
			
			select trans
			skip
		Enddo
		select collect
		replace posted with 'P'
		
		select tmppost
		skip
Enddo
select collect
=tableupdate(1,.t.)

select obil_mst
=tableupdate(1,.t.)
use

select obil_det
=tableupdate(1,.t.)

thisform.release

ENDPROC

PROCEDURE Valid
select collect
tmpdate = this.value
if used("tmppost")
	use in tmppost
endif
set safety off
if !empty(tmpdate)
	select date,indentno,name,regno,total,posted from collect ;
	where date=tmpdate and empty(posted) and left(type,1) = 'O' ;
		and !empty(regno) order by date,indentno into table tmppost
else
	select date,indentno,name,regno,total,posted from collect ;
	where empty(posted) and left(type,1) = 'O' ;
		and !empty(regno) order by date,indentno into table tmppost
endif
set safety on
thisform.listgrid1.recordsource = "tmppost"
thisform.listgrid1.column1.width=57

thisform.listgrid1.column6.width=46
thisform.listgrid1.readonly=.t.
thisform.listgrid1.refresh

ENDPROC

PROCEDURE gettext
para txt
TXTLEN= LEN(TXT)
append blan in prn
** setting default values
repl tul with .f.,tbold with .f.,talign with 'L'
**... format
** starting position of format codes in the  string
start = at('\p',txt)
if substr(txt,start,5) = '\pard' or substr(txt,start,6) = '\plain'
	nchar = at(space(1),right(txt,txtlen-start))
	tformat = substr(txt,start,nchar)
else
	if substr(txt,start+5,5) = '\pard' or substr(txt,start+5,6) = '\plain'
		start = start + 5
		nchar = at(space(1),right(txt,txtlen-start))
		tformat = substr(txt,start,nchar)
	else
		nchar   = at(space(1),right(txt,txtlen-start))
		tformat = this.tformat
	Endif
endif		
** checking font bold
if at("\b",tformat)!= 0
		repl prn.tbold with .t.
Endif
** checking font underline
if at("\ul",tformat)!= 0
	repl prn.tul with .t.
endif

** checking of alligment

do case
	case at("qc",tformat)!=0
		repl prn.talign with 'C'
	case at("qr",tformat)!=0
		repl prn.talign with 'R'
	other 
		repl prn.talign with 'L'
Endcase

** getting end of text
textend= at('\p',right(txt,txtlen-(start+nchar))) -1
** Setting report text
this.tformat = tformat 
tmptext = substr(txt,start+nchar+1,textend)
** skipping Additonal Control Character
pos = 1
do while pos!=0
	pos = at('\',tmptext)
	if pos!=0
		pos2 = at(' ',right(tmptext,len(tmptext) - pos) )
		tmptext = left(tmptext,pos -1)+ ' ' +right(tmptext,len(tmptext)- (pos+pos2) )
	endif	
Enddo	
repl prn.text with tmptext &&substr(txt,start+nchar+1,tmptext)
return start + nchar + textend
***  AT(cSearchExpression, cExpressionSearched [, nOccurrence])
***  SUBSTR(cExpression, nStartPosition [, nCharactersReturned])  

ENDPROC

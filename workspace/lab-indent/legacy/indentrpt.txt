PROCEDURE repoquery
select * from collect where collect.indentno == thisform.txtindent.value into cursor tmpind

select * from trans where trans.indentno == thisform.txtindent.value into cursor tmptran

select * ;
	from tmpind, tmptran ;
	where tmpind.indentno == tmptran.indentno ;
		into cursor tmpschd

*select * ;
*	from report, testtr ;
*	where report.labno == thisform.txtindent.value and ;
*		report.labno == testtr.labno ;
*		into cursor tmpschd
	
if thisform.printflag
	report form indentr to print noconsole
else
	report form indentr preview 
endif

use in tmpschd

ENDPROC
